<?php
// Chat Bot - AI Responses Only with Voice Chat & Admin Management

class angleChatBot {
    private $telegram_token = '8954943501:AAETwSNxNEEcIDz2VyqMuP-pw_wClU7q5W0';
    private $webhook_url;
    
    private $persona = [
        'name' => 'Angle', 'gender' => 'female', 'age' => '21',
        'location' => 'Bihar', 'traits' => 'friendly, playful, witty'
    ];
    
    private $api_urls = [
        'primary' => 'https://workers.dev',
        'backup' => 'https://workers.dev'
    ];
    
    private $owners = []; 
    private $bot_username = 'Angle_Chat_Super_bot';
    private $db_file = 'bot_database.json';
    
    private $abuse_words = [
        'bhosdi', 'bhosad', 'madarchod', 'maderchod', 'mc', 'bc', 'benchod', 'behenchod',
        'gaand', 'gand', 'chutiya', 'chut', 'lund', 'loda', 'lavde', 'lavda',
        'randi', 'bhenchod', 'motherfucker', 'fuck', 'shit', 'asshole', 'bitch'
    ];
    
    private $blocked_error_patterns = [
        '/an error occurred/i', '/error occurred/i', '/please try again/i',
        '/something went wrong/i', '/internal server error/i'
    ];

    public function __construct() {
        $this->webhook_url = $_ENV['WEBHOOK_URL'] ?? 'https://onrender.com';
        if (!file_exists($this->db_file)) {
            $initial_data = [
                'groups' => [], 'stats' => ['total_messages' => 0], 
                'banned' => [], 'users' => [], 'admins' => [], 'voice_chat_active' => []
            ];
            file_put_contents($this->db_file, json_encode($initial_data, JSON_PRETTY_PRINT));
        }
    }

    private function getDatabase() {
        if (!file_exists($this->db_file)) return [];
        return json_decode(file_get_contents($this->db_file), true) ?? [];
    }

    private function saveDatabase($data) {
        file_put_contents($this->db_file, json_encode($data, JSON_PRETTY_PRINT));
    }
    
    private function isAdmin($user_id) {
        $db = $this->getDatabase();
        return in_array($user_id, $this->owners) || in_array($user_id, $db['admins'] ?? []);
    }
    
    private function isOwner($user_id) {
        return in_array($user_id, $this->owners);
    }
    
    private function addAdmin($user_id) {
        if ($this->isOwner($user_id)) return false;
        $db = $this->getDatabase();
        if (!isset($db['admins'])) $db['admins'] = [];
        if (!in_array($user_id, $db['admins'])) {
            $db['admins'][] = $user_id; $this->saveDatabase($db); return true;
        }
        return false;
    }
    
    private function removeAdmin($user_id) {
        if ($this->isOwner($user_id)) return false;
        $db = $this->getDatabase();
        if (isset($db['admins']) && in_array($user_id, $db['admins'])) {
            $db['admins'] = array_values(array_diff($db['admins'], [$user_id]));
            $this->saveDatabase($db); return true;
        }
        return false;
    }
    
    private function getAdminList() {
        $db = $this->getDatabase();
        $admins = array_unique(array_merge($this->owners, $db['admins'] ?? []));
        sort($admins); return $admins;
    }
    
    private function containsAbuse($message) {
        $message_lower = strtolower($message);
        foreach ($this->abuse_words as $word) {
            if (strpos($message_lower, $word) !== false) return true;
        }
        return false;
    }
    
    private function isErrorMessage($text) {
        foreach ($this->blocked_error_patterns as $pattern) {
            if (preg_match($pattern, $text)) return true;
        }
        return false;
    }
    
    private function sendReaction($chat_id, $message_id, $emoji) {
        $url = "https://telegram.org{$this->telegram_token}/setMessageReaction";
        $data = [
            'chat_id' => $chat_id, 'message_id' => $message_id,
            'reaction' => json_encode([['type' => 'emoji', 'emoji' => $emoji]])
        ];
        $this->sendRequest($url, $data);
    }
    
    private function getRandomReaction($message) {
        if (preg_match('/love|like|❤️|pyaar/i', $message)) return '❤️';
        if (preg_match('/lol|laugh|😂/i', $message)) return '😂';
        if (preg_match('/sad|😢/i', $message)) return '😢';
        if (rand(1, 100) <= 20) {
            $reactions = ['👍', '❤️', '🔥', '🥰', '👏', '😁'];
            return $reactions[array_rand($reactions)];
        }
        return '';
    }
    
    private function isBotMentioned($message) {
        if (preg_match('/@' . preg_quote($this->bot_username, '/') . '/i', $message)) return true;
        foreach (['/^bot\s/i', '/^angle\s/i', '/\sbot[?\s,!.]/i'] as $pattern) {
            if (preg_match($pattern, $message)) return true;
        }
        return false;
    }
    
    private function cleanMessage($message) {
        $message = preg_replace('/@' . preg_quote($this->bot_username, '/') . '/i', '', $message);
        return trim(preg_replace('/^(bot|angle)[\s,.:;!?]*/i', '', $message));
    }

    public function processMessage($update) {
        if (!isset($update['message'])) return;
        $msg_data = $update['message'];
        $message = $msg_data['text'] ?? '';
        $chat_id = $msg_data['chat']['id'] ?? 0;
        $chat_type = $msg_data['chat']['type'] ?? '';
        $user_id = $msg_data['from']['id'] ?? 0;
        $first_name = $msg_data['from']['first_name'] ?? 'There';
        $message_id = $msg_data['message_id'] ?? null;
        
        if ($user_id <= 0) return;
        $db = $this->getDatabase();
        if (in_array($user_id, $db['banned'] ?? [])) return;
        
        $db['users'][$user_id] = ['first_name' => $first_name, 'last_interaction' => date('Y-m-d H:i:s')];
        $this->saveDatabase($db);
        
        if (isset($msg_data['video_chat_started'])) {
            $this->sendMessage($chat_id, "<b>🎙️ Voice chat started</b>", $message_id);
            return;
        }

        if (isset($msg_data['new_chat_members'])) {
            foreach ($msg_data['new_chat_members'] as $m) {
                if ($m['username'] !== $this->bot_username) {
                    $this->sendMessage($chat_id, "<b>Welcome 🎉</b>", $message_id);
                }
            }
            return;
        }

        if (isset($msg_data['left_chat_member'])) {
            $left_member = $msg_data['left_chat_member'];
            if ($left_member['username'] !== $this->bot_username) {
                $this->sendMessage($chat_id, "<b>Good Bye 👋</b>", $message_id);
            }
            return;
        }

        if (substr($message, 0, 1) === '/') {
            $this->handleCommand($chat_id, $message, $user_id, $message_id);
            return;
        }

        $is_abuse = $this->containsAbuse($message);
        $should_respond = ($chat_type === 'private' || $is_abuse);
        if (strpos($chat_type, 'group') !== false) {
            if ($this->isBotMentioned($message) || (isset($msg_data['reply_to_message']['from']) && $msg_data['reply_to_message']['from']['username'] == $this->bot_username)) {
                $should_respond = true; $message = $this->cleanMessage($message);
            }
        }

        if ($should_respond && !empty($message)) {
            $this->sendChatAction($chat_id, 'typing');
            $response = $this->getAIResponse($message, $first_name, $is_abuse);
            $this->sendMessage($chat_id, $response, $message_id);
        }
    }

    private function getBotId() {
        $url = "https://telegram.org{$this->telegram_token}/getMe";
        $result = @file_get_contents($url);
        if ($result) {
            $data = json_decode($result, true);
            return $data['result']['id'] ?? 0;
        }
        return 0;
    }

    private function handleCommand($chat_id, $message, $user_id, $reply_id) {
        $parts = explode(' ', $message); $command = strtolower($parts[0]);
        if ($command === '/start') {
            $this->sendMessage($chat_id, "Hey, I'm angle", $reply_id);
        }
    }

    private function sendRequest($url, $data) {
        $options = ['http' => ['header' => "Content-type: application/x-www-form-urlencoded\r\n", 'method' => 'POST', 'content' => http_build_query($data), 'timeout' => 10]];
        return @file_get_contents($url, false, stream_context_create($options));
    }

    private function getAIResponse($message, $user_name, $is_abuse) {
        $prefix = "You are angle a 21 year old Indian girl from Delhi. IMPORTANT: RESPONSE MUST BE VERY SHORT - Maximum 5-7 words only. No punctuation, no emojis. ";
        $prompt = $is_abuse ? $prefix . "User abused you: \"{$message}\". Reply aggressively:" : $prefix . "User message: \"{$message}\". Reply friendly:";
        
        foreach ([$this->api_urls['primary'], $this->api_urls['backup']] as $url) {
            try {
                $res = @file_get_contents($url . urlencode($prompt), false, stream_context_create(['http' => ['timeout' => 5]]));
                if ($res) {
                    $data = json_decode($res, true);
                    $text = $data['message'] ?? $data['response'] ?? '';
                    if ($text && !$this->isErrorMessage($text)) return $this->cleanResponse($text);
                }
            } catch (Exception $e) {}
        }
        return $is_abuse ? "Tujhe kya problem hai" : "Haan ji bolo";
    }

    private function cleanResponse($text) {
        $text = preg_replace('/[\x{1F600}-\x{1F64F}|\x{2600}-\x{27BF}]/u', '', $text);
        $words = explode(' ', trim(preg_replace('/\s+/', ' ', $text)));
        return implode(' ', array_slice($words, 0, 7));
    }

    private function sendChatAction($chat_id, $action) { 
        $this->sendRequest("https://telegram.org{$this->telegram_token}/sendChatAction", ['chat_id' => $chat_id, 'action' => $action]); 
    }
    
    private function sendMessage($chat_id, $text, $reply_id = null) {

$data = ['chat_id' => $chat_id, 'text' => $text, 'parse_mode' => 'HTML'];
if ($reply_id) $data['reply_to_message_id'] = $reply_id;
return $this->sendRequest("telegram.org{$this->telegram_token}/sendMessage", $data);
}
public function handleWebhook() {
$update = json_decode(file_get_contents('php://input'), true);
if ($update) $this->processMessage($update);
http_response_code(200); echo "OK";
}
}
$bot = new angleChatBot();
$bot->handleWebhook();
