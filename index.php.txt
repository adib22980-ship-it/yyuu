    public function __construct() {

    private function getDatabase() {
        if (!file_exists($this->db_file)) return [];
        return json_decode(file_get_contents($this->db_file), true) ?? [];
    }

    private function saveDatabase($data) {
        file_put_contents($this->db_file, json_encode($data, JSON_PRETTY_PRINT));
    }
    
    private function isAdmin($user_id) {
        $db = $this->getDatabase();
        $admins = $db['admins'] ?? [];
        if (in_array($user_id, $this->owners)) return true;
        return in_array($user_id, $admins);
    }
    
    private function isOwner($user_id) {
        return in_array($user_id, $this->owners);
    }
    
    private function addAdmin($user_id) {
        if ($this->isOwner($user_id)) return false;
        $db = $this->getDatabase();
        if (!isset($db['admins'])) $db['admins'] = [];
        if (!in_array($user_id, $db['admins'])) {
            $db['admins'][] = $user_id;
            $this->saveDatabase($db);
            return true;
        }
        return false;
    }
    
    private function removeAdmin($user_id) {
        if ($this->isOwner($user_id)) return false;
        $db = $this->getDatabase();
        if (isset($db['admins']) && in_array($user_id, $db['admins'])) {
            $db['admins'] = array_values(array_diff($db['admins'], [$user_id]));
            $this->saveDatabase($db);
            return true;
        }
        return false;
    }
    
    private function getAdminList() {
        $db = $this->getDatabase();

    private function containsAbuse($message) {

        foreach ($this->abuse_words as $word) {

    }

    private function isBotMentioned($message) {
        if (preg_match('/@' . preg_quote($this->bot_username, '/') . '/i', $message)) return true;
        $patterns = ['/^bot\s/i', '/^angle\s/i', '/\sbot[?\s,!.]/i', '/\sangle[?\s,!.]/i'];
        foreach ($patterns as $pattern) {
            if (preg_match($pattern, $message)) return true;
        }
        return false;
    }
    
    private function cleanMessage($message) {
        $message = preg_replace('/@' . preg_quote($this->bot_username, '/') . '/i', '', $message);
        $message = preg_replace('/^(bot|angle)[\s,.:;!?]*/i', '', $message);
        return trim($message);
    }

        if (!isset($update['message'])) return;

        $username = $user['username'] ?? '';
        $message_id = $msg_data['message_id'] ?? null;
        $reply_to_message_id = $message_id;
        
        if ($user_id <= 0) return;

                'last_interaction' => date('Y-m-d H:i:s')

            }
